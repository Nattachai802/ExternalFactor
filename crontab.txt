# exFactor — cron jobs (เวลาไทย)
#
# ติดตั้ง:   crontab crontab.txt
# ดูของเดิม: crontab -l
# ลบทั้งหมด: crontab -r
# ตรวจก่อนติดตั้ง: python check_cron.py        (ดูรอบถัดไป + ตรวจคำสั่ง)
#                  python check_cron.py --run  (รันจริงด้วย env แบบ cron)
#
# ⚠️ ก่อนติดตั้ง: cron ไม่อ่าน .env ให้ — jobs.py เรียก load_dotenv() เองแล้ว
#    แต่ต้อง cd เข้า project ก่อนเสมอ ไม่งั้นหา .env ไม่เจอ (ทุกบรรทัดจึงมี cd นำ)
#
# ⚠️ เครื่อง server ต้องตั้ง timezone เป็น Asia/Bangkok ไม่งั้นเวลาเพี้ยน 7 ชม.
#    เช็ค: timedatectl        ตั้ง: sudo timedatectl set-timezone Asia/Bangkok
#    (CRON_TZ ใช้ได้บน Linux cron/cronie แต่ macOS ไม่รองรับ — อย่าพึ่งบรรทัดนั้นอย่างเดียว)
#
# log แยกไฟล์ต่องาน เก็บทั้ง stdout และ stderr (2>&1) — งานพังจะเห็น traceback เต็ม
# jobs.py exit 1 เมื่อพัง cron จะส่งเมลถ้าตั้ง MAILTO ไว้

SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin
CRON_TZ=Asia/Bangkok

# ── รายวัน (ช่วงตี 1) ────────────────────────────────────────
# ราคาน้ำมันประกาศเช้าตรู่ — ดึงตอนตี 1 หลังประกาศแล้ว
0  1 * * *  cd /Users/rzhun/ex-factor && .venv/bin/python jobs.py energy       >> output/cron_energy.log 2>&1

# ปฏิทินจีน — ต้องมีข้อมูลก่อนร้านเปิด
15 1 * * *  cd /Users/rzhun/ex-factor && .venv/bin/python jobs.py myth         >> output/cron_myth.log 2>&1

# ราคาวัตถุดิบ DIT — ดึงย้อนหลัง 7 วันกันวันที่ประกาศช้า/วันหยุดไม่มีประกาศ
30 1 * * *  cd /Users/rzhun/ex-factor && .venv/bin/python jobs.py food_price --days 7 >> output/cron_food_price.log 2>&1

# ── รายสัปดาห์ (จันทร์) ──────────────────────────────────────
# อัตราแลกเปลี่ยน + เงินเฟ้อ — เงินเฟ้อประกาศรายเดือน อัตราแลกเปลี่ยนขยับรายวัน
# แต่ใช้ประกอบการตัดสินใจระดับสัปดาห์ ไม่ต้องละเอียดรายวัน
0  2 * * 1  cd /Users/rzhun/ex-factor && .venv/bin/python jobs.py economic     >> output/cron_economic.log 2>&1

# ── รายเดือน (วันที่ 1) ──────────────────────────────────────
# ค่าไฟ/ค่าแรง เปลี่ยนปีละไม่กี่ครั้ง — เช็คเดือนละครั้งพอ
0  3 1 * *  cd /Users/rzhun/ex-factor && .venv/bin/python jobs.py electricity  >> output/cron_electricity.log 2>&1
15 3 1 * *  cd /Users/rzhun/ex-factor && .venv/bin/python jobs.py wage         >> output/cron_wage.log 2>&1

# ── ทุกครึ่งปี (1 ม.ค. และ 1 ก.ค.) ───────────────────────────
# สีเสื้อมงคล — ตารางคงที่ผูกวันในสัปดาห์ ไม่เปลี่ยนตามวันจริง รันไว้กันแถวหายเฉยๆ
0  4 1 1,7 *  cd /Users/rzhun/ex-factor && .venv/bin/python jobs.py lucky_shirt  >> output/cron_lucky_shirt.log 2>&1

# วันหยุด/เทศกาล — ทุกครึ่งปีจับวันหยุดพิเศษที่ ครม.ประกาศแทรกกลางปีได้ไวขึ้นกว่ารายปี
# ⚠️ ยังมีช่วงว่างได้ถึง ~6 เดือน ถ้าประกาศมากะทันหันจริงๆ ต้องรันมือ: python jobs.py holiday
15 4 1 1,7 *  cd /Users/rzhun/ex-factor && .venv/bin/python jobs.py holiday      >> output/cron_holiday.log 2>&1
